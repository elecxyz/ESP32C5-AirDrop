cmake_minimum_required(VERSION 3.16)
project(cachydrop_c5_host C)
set(CMAKE_C_STANDARD 11)
get_filename_component(OWL "${CMAKE_CURRENT_SOURCE_DIR}/../../owl" ABSOLUTE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NL REQUIRED IMPORTED_TARGET libnl-3.0 libnl-genl-3.0 libnl-route-3.0)
pkg_check_modules(PCAP REQUIRED IMPORTED_TARGET libpcap)
find_package(ZLIB REQUIRED)
find_library(EV ev REQUIRED)
file(GLOB AWDL_SOURCES CONFIGURE_DEPENDS "${OWL}/src/*.c")
add_library(c5_awdl STATIC ${AWDL_SOURCES} "${OWL}/radiotap/radiotap.c")
target_include_directories(c5_awdl PUBLIC "${OWL}/src" "${OWL}/radiotap")
target_compile_definitions(c5_awdl PRIVATE _DEFAULT_SOURCE RADIOTAP_SUPPORT_OVERRIDES crc32=owl_crc32)
add_executable(cachydrop-c5 main.c transport.c "${OWL}/daemon/core.c" "${OWL}/daemon/netutils.c")
target_include_directories(cachydrop-c5 PRIVATE "${OWL}/daemon")
# Preserve the pinned OWL submodule. Only this experimental binary replaces
# libpcap dispatch with a serial transport; all other libpcap calls stay native.
set_source_files_properties("${OWL}/daemon/core.c" PROPERTIES COMPILE_DEFINITIONS "pcap_dispatch=c5_dispatch")
target_compile_options(cachydrop-c5 PRIVATE -Wall -Wextra)
target_link_options(cachydrop-c5 PRIVATE "-Wl,--wrap=get_hostname")
target_link_libraries(cachydrop-c5 PRIVATE c5_awdl PkgConfig::NL PkgConfig::PCAP ZLIB::ZLIB ${EV})
